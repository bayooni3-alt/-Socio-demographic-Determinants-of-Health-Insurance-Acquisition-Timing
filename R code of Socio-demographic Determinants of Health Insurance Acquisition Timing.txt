# ============================================
# STEP 1: CREATE A CLEAN DATASET (NO MISSING VALUES)
# ============================================

# Load data
df <- read.csv("C:/Users/USER/Documents/HealthInsurance.csv")

# Remove rownames if exists
if("rownames" %in% colnames(df)) df$rownames <- NULL

# Create clean dataset with NO missing values
df_clean <- df %>%
  mutate(event = ifelse(insurance == "yes", 1, 0)) %>%
  select(age, gender, married, selfemp, family, region, education, health, event) %>%
  na.omit() %>%
  mutate(across(where(is.character), as.factor))

cat("Clean dataset size:", nrow(df_clean), "rows\n")
cat("Events (insurance acquired):", sum(df_clean$event), "\n")
cat("Censored (no insurance):", sum(df_clean$event == 0), "\n\n")

# Create survival object
surv_clean <- Surv(time = df_clean$age, event = df_clean$event)

# ============================================
# STEP 2: FIT ALL MODELS USING THE SAME DATASET
# ============================================

cat("========== FITTING MODELS ==========\n")

# Model 1: Null model
cox_null <- coxph(surv_clean ~ 1, data = df_clean)

# Model 2: Minimal model
cox_minimal <- coxph(surv_clean ~ gender + married, data = df_clean)

# Model 3: Medium model
cox_medium <- coxph(surv_clean ~ gender + married + selfemp + family, data = df_clean)

# Model 4: Full model (no age - it's the time scale!)
cox_full <- coxph(surv_clean ~ gender + married + selfemp + family + 
                   region + education + health, data = df_clean)

# Model 5: Stratified by region
cox_stratified <- coxph(surv_clean ~ gender + married + selfemp + 
                         family + education + health + strata(region), 
                         data = df_clean)

# Model 6: Time-varying effect for family size (ONLY continuous variable)
cox_tvc_family <- coxph(surv_clean ~ gender + married + selfemp +
                         family + tt(family) + region + education + health,
                         data = df_clean,
                         tt = function(x, t, ...) x * log(t))

# Model 7: Stratified by age period
df_clean$age_period <- cut(df_clean$age, 
                            breaks = c(18, 40, 60, 100),
                            labels = c("18-40", "41-60", "61+"))

cox_stratified_age <- coxph(surv_clean ~ gender + married + selfemp + 
                             family + region + education + health + 
                             strata(age_period), data = df_clean)

cat("✓ All 7 models fitted successfully on", nrow(df_clean), "observations\n\n")

# STEP 4: LOG-RANK TESTS
# ============================================

cat("========== LOG-RANK TESTS ==========\n")

logrank_gender <- survdiff(surv_clean ~ gender, data = df_clean)
logrank_married <- survdiff(surv_clean ~ married, data = df_clean)
logrank_selfemp <- survdiff(surv_clean ~ selfemp, data = df_clean)
logrank_region <- survdiff(surv_clean ~ region, data = df_clean)
logrank_education <- survdiff(surv_clean ~ education, data = df_clean)

logrank_results <- data.frame(
  Variable = c("Gender", "Married", "Self-Employed", "Region", "Education"),
  Chi_Square = c(logrank_gender$chisq, logrank_married$chisq, logrank_selfemp$chisq,
                 logrank_region$chisq, logrank_education$chisq),
  df = c(length(logrank_gender$n) - 1, length(logrank_married$n) - 1,
         length(logrank_selfemp$n) - 1, length(logrank_region$n) - 1,
         length(logrank_education$n) - 1),
  P_value = c(1 - pchisq(logrank_gender$chisq, length(logrank_gender$n) - 1),
              1 - pchisq(logrank_married$chisq, length(logrank_married$n) - 1),
              1 - pchisq(logrank_selfemp$chisq, length(logrank_selfemp$n) - 1),
              1 - pchisq(logrank_region$chisq, length(logrank_region$n) - 1),
              1 - pchisq(logrank_education$chisq, length(logrank_education$n) - 1))
)

# ============================================
# STEP 5: MODEL COMPARISONS
# ============================================

cat("========== MODEL COMPARISONS ==========\n")

# Compare nested models
cat("1. Null vs Minimal vs Medium vs Full:\n")
anova_nested <- anova(cox_null, cox_minimal, cox_medium, cox_full, test = "Chisq")
print(anova_nested)

# Compare Full vs Time-varying family using manual likelihood ratio test
cat("2. Full model vs Time-varying family model:\n")
loglik_full <- as.numeric(logLik(cox_full))
loglik_tvc <- as.numeric(logLik(cox_tvc_family))
lr_stat <- -2 * (loglik_full - loglik_tvc)
lr_df <- length(coef(cox_tvc_family)) - length(coef(cox_full))
lr_pvalue <- 1 - pchisq(lr_stat, lr_df)

cat("Likelihood ratio test statistic:", round(lr_stat, 2), "\n")
cat("Degrees of freedom:", lr_df, "\n")
cat("P-value:", format.pval(lr_pvalue, digits = 4), "\n\n")

# AIC comparison
cat("3. AIC Comparison (all models):\n")
aic_table <- data.frame(
  Model = c("Null", "Minimal", "Medium", "Full", "Stratified (Region)", 
            "Time-Varying Family", "Stratified (Age)"),
  AIC = c(AIC(cox_null), AIC(cox_minimal), AIC(cox_medium), 
          AIC(cox_full), AIC(cox_stratified), 
          AIC(cox_tvc_family), AIC(cox_stratified_age))
)

aic_table <- aic_table[order(aic_table$AIC), ]

# STEP 10: SURVIVAL PREDICTIONS FOR RISK PROFILES
# ============================================

cat("========== SURVIVAL PREDICTIONS ==========\n")

# Define risk profiles - ensure factor levels match df_clean
profiles <- data.frame(
  profile = c("Low Risk", "Medium Risk", "High Risk"),
  gender = factor(c("female", "male", "male"), levels = levels(df_clean$gender)),
  married = factor(c("yes", "yes", "no"), levels = levels(df_clean$married)),
  selfemp = factor(c("no", "no", "yes"), levels = levels(df_clean$selfemp)),
  family = c(2, 4, 6),
  region = factor(c("south", "south", "south"), levels = levels(df_clean$region)),
  education = factor(c("master", "bachelor", "none"), levels = levels(df_clean$education)),
  health = factor(c("yes", "yes", "no"), levels = levels(df_clean$health))
)

print(profiles)

# Fit Weibull model
weibull_pred <- flexsurvreg(surv_clean ~ gender + married + selfemp + 
                             family + region + education + health,
                             data = df_clean, dist = "weibull")

cat("Weibull model fitted successfully\n")

# Predict survival
age_seq <- seq(20, 80, by = 5)

# Use tryCatch for safe prediction
pred_curves <- tryCatch({
  summary(weibull_pred, newdata = profiles, t = age_seq, type = "survival")
}, error = function(e) {
  cat("Warning:", e$message, "\n")
  cat("Using manual prediction method...\n")
  
  # Manual prediction
  coef_w <- coef(weibull_pred)
  pred_mat <- matrix(NA, nrow = length(age_seq), ncol = nrow(profiles))
  
  for(i in 1:nrow(profiles)) {
    # Calculate linear predictor
    lp <- 0
    if(profiles$gender[i] == "male") lp <- lp + coef_w["gendermale"]
    if(profiles$married[i] == "yes") lp <- lp + coef_w["marriedyes"]
    if(profiles$selfemp[i] == "yes") lp <- lp + coef_w["selfempyes"]
    lp <- lp + profiles$family[i] * coef_w["family"]
    if(profiles$region[i] == "northeast") lp <- lp + coef_w["regionnortheast"]
    if(profiles$region[i] == "south") lp <- lp + coef_w["regionsouth"]
    if(profiles$region[i] == "west") lp <- lp + coef_w["regionwest"]
    if(profiles$education[i] == "ged") lp <- lp + coef_w["educationged"]
    if(profiles$education[i] == "highschool") lp <- lp + coef_w["educationhighschool"]
    if(profiles$education[i] == "master") lp <- lp + coef_w["educationmaster"]
    if(profiles$education[i] == "none") lp <- lp + coef_w["educationnone"]
    if(profiles$education[i] == "other") lp <- lp + coef_w["educationother"]
    if(profiles$education[i] == "phd") lp <- lp + coef_w["educationphd"]
    if(profiles$health[i] == "yes") lp <- lp + coef_w["healthyes"]
    
    # Weibull survival
    shape <- exp(coef_w["shape"])
    scale <- exp(-lp/shape)
    pred_mat[, i] <- exp(- (age_seq / scale)^shape)
  }
  
  return(list(est = pred_mat))
})

# Create prediction plot

cat("========== 5-FOLD CROSS-VALIDATION ==========\n")

set.seed(456)

cv_results <- data.frame(Fold = 1:k, C_index = NA)

for(i in 1:k) {
  train_data <- df_clean[folds != i, ]
  test_data <- df_clean[folds == i, ]
  
  model_cv <- coxph(Surv(age, event) ~ gender + married + selfemp + 
                     family + region + education + health,
                     data = train_data)
  
  lp <- predict(model_cv, newdata = test_data, type = "lp")
  test_surv <- Surv(test_data$age, test_data$event)
  c_index <- concordance(test_surv ~ lp)$concordance
  cv_results$C_index[i] <- c_index
  
  cat("Fold", i, "- C-index:", round(c_index, 4), "\n")
}

cat("\nAverage C-index:", round(mean(cv_results$C_index), 4), "\n")
cat("Standard deviation:", round(sd(cv_results$C_index), 4), "\n\n")

boot_hr <- function(data, indices) {

  model <- coxph(Surv(age, event) ~ married + selfemp + family, data = d)
